    PostgreSQL バージョン・ステータス確認

    サーバーのバージョン、起動時刻、アップタイムを確認
    サーバーのバージョン、起動時刻、経過時間を確認します。サーバーの稼働状況を把握するための最初のチェック項目です。 
    SELECT
    version() AS version,
    pg_postmaster_start_time() AS start_time,
    age(now(), pg_postmaster_start_time()) AS uptime,
    current_database() AS database,
    current_user AS user;

    設定パラメータの確認 基本
    サーバー、クラスタ、ディレクトリ、ポート、メモリ、WAL、並列処理、プリロードライブラリなどの主要設定を表示します。構成の確認と推奨値との比較に使用します。 
    SELECT name, setting, unit, source
FROM pg_settings
WHERE name IN (
    'server_version', 'server_version_num', 'cluster_name', 'data_directory',
    'config_file', 'hba_file', 'port', 'listen_addresses', 'max_connections',
    'superuser_reserved_connections', 'shared_buffers', 'work_mem',
    'maintenance_work_mem', 'effective_cache_size', 'huge_pages',
    'max_worker_processes', 'max_parallel_workers', 'max_parallel_workers_per_gather',
    'max_parallel_maintenance_workers', 'shared_preload_libraries'
)
ORDER BY name;

    アクティブセッション・接続状況 重要
    全セッション（アクティブ・アイドル・その他）の詳細を表示します。PID、ユーザー、データベース、状態、実行クエリ、トランザクション開始時刻、待機イベントなどを確認します。問題診断の最初の段階で必ず確認してください。 
    SELECT
    pid, usename, datname, application_name, client_addr,
    backend_type, state, wait_event_type, wait_event,
    xact_start, query_start, state_change,
    now() - xact_start  AS xact_age,
    now() - query_start AS query_age,
    backend_xid, backend_xmin,
    left(query, 200) AS query
FROM pg_stat_activity
ORDER BY xact_start NULLS LAST;

    データベースサイズ・接続数 基本
    各データベースのサイズ、接続数、最後のVacuum/Analyze実行時刻を表示します。ディスク容量計画とメンテナンス状況の確認に使用します。 
    SELECT
    datname AS database,
    pg_size_pretty(pg_database_size(datname)) AS size,
    numbackends AS connections,
    datconnlimit AS max_connections,
    last_vacuum, last_autovacuum, last_analyze, last_autoanalyze
FROM pg_stat_database
WHERE datname NOT IN ('postgres', 'template0', 'template1')
ORDER BY pg_database_size(datname) DESC;

    パフォーマンス・キャッシュ統計

    キャッシュ・ヒット率（集計） 性能
    全Heapテーブルと全Indexの合計キャッシュ・ヒット率を表示します。99%以上が目標です。この値が低い場合は shared_buffers を増やすか、アクセスパターンを最適化する必要があります。 
    SELECT
    'Heap' AS type,
    ROUND(100 * heap_blks_hit /
        NULLIF(heap_blks_hit + heap_blks_read, 0)::numeric, 2)
        AS hit_ratio_percent
FROM pg_statio_user_tables
WHERE heap_blks_hit + heap_blks_read > 0
UNION ALL
SELECT
    'Index',
    ROUND(100 * idx_blks_hit /
        NULLIF(idx_blks_hit + idx_blks_read, 0)::numeric, 2)
FROM pg_statio_user_indexes
WHERE idx_blks_hit + idx_blks_read > 0
ORDER BY type;

    テーブル毎のキャッシュ・ヒット率 性能
    テーブル単位でのHeapキャッシュ・ヒット率を計算します。アクセス多いテーブルの中でヒット率が低いものを特定できます。95%未満のテーブルはメモリ不足またはフルテーブルスキャン多い可能性があります。 
    SELECT
    schemaname, relname, heap_blks_hit, heap_blks_read,
    CASE WHEN heap_blks_hit + heap_blks_read = 0 THEN NULL
         ELSE ROUND(100 * heap_blks_hit /
              (heap_blks_hit + heap_blks_read)::numeric, 2)
         END AS heap_hit_ratio_percent,
    heap_blks_hit + heap_blks_read AS total_heap_blocks_accessed
FROM pg_statio_user_tables
WHERE heap_blks_hit + heap_blks_read > 10
ORDER BY heap_hit_ratio_percent ASC NULLS LAST
LIMIT 30;

    インデックス毎のキャッシュ・ヒット率 性能
    インデックス単位でのキャッシュ・ヒット率を表示します。ヒット率が低いインデックスはサイズが大きすぎるか、使用頻度が低い可能性があります。 
    SELECT
    schemaname, relname, indexrelname, idx_blks_hit, idx_blks_read,
    CASE WHEN idx_blks_hit + idx_blks_read = 0 THEN NULL
         ELSE ROUND(100 * idx_blks_hit /
              (idx_blks_hit + idx_blks_read)::numeric, 2)
         END AS idx_hit_ratio_percent,
    idx_blks_hit + idx_blks_read AS total_idx_blocks_accessed
FROM pg_statio_user_indexes
WHERE idx_blks_hit + idx_blks_read > 10
ORDER BY idx_hit_ratio_percent ASC NULLS LAST
LIMIT 30;

    テーブル・インデックス アクセス統計 性能
    各テーブルのseq_scan vs index_scan比率、insert/update/delete数を表示します。seq_scan比率が高い場合はIndexが不足している可能性があります。また、未使用のインデックスも特定できます。 
    -- テーブルアクセス統計（seq_scan vs index_scan）
SELECT
    schemaname, relname, seq_scan, idx_scan,
    CASE WHEN seq_scan + idx_scan > 0
         THEN ROUND(100.0 * seq_scan / (seq_scan + idx_scan), 2)
         ELSE 0 END AS seq_scan_ratio_percent,
    n_tup_ins, n_tup_upd, n_tup_del,
    last_vacuum, last_autovacuum
FROM pg_stat_user_tables
ORDER BY seq_scan DESC
LIMIT 20;

-- インデックス使用統計
SELECT
    schemaname, relname, indexrelname,
    idx_scan, idx_tup_read, idx_tup_fetch
FROM pg_stat_user_indexes
ORDER BY idx_scan NULLS FIRST;

-- 未使用インデックス候補（メモリ・ディスク浪費の可能性）
SELECT
    schemaname, relname, indexrelname, idx_scan,
    pg_size_pretty(pg_relation_size(indexrelid)) AS index_size
FROM pg_stat_user_indexes
WHERE idx_scan = 0
ORDER BY pg_relation_size(indexrelid) DESC;

    I/O・ストレージ・ディスク監視

    I/O統計詳細（PG18新機能)
    PostgreSQL 18で追加された pg_stat_io ビューでバイト単位の I/O ボリュームを直接監視できます。read_bytes、write_bytes、extend_bytesで実際の I/O 負荷を把握します。 
    -- I/O統計の詳細（backend_type別、object別、context別）
SELECT
    backend_type, object, context,
    reads, read_bytes, read_time,
    writes, write_bytes, write_time,
    writebacks, writeback_time,
    extends, extend_bytes, extend_time,
    fsyncs, fsync_time, evictions
FROM pg_stat_io
ORDER BY backend_type, object, context;

-- I/O統計の集計（backend_typeとobjectごと）
SELECT
    backend_type, object,
    sum(reads) AS reads,
    sum(read_bytes) AS read_bytes,
    sum(writes) AS writes,
    sum(write_bytes) AS write_bytes,
    sum(fsyncs) AS fsyncs,
    sum(evictions) AS evictions
FROM pg_stat_io
GROUP BY backend_type, object
ORDER BY read_bytes DESC;

    WAL統計 レプリケーション
    WALレコード生成数、Full Page Image（FPI）数、WALバイト数、FSyncの呼び出し数などを表示します。WAL生成のI/O負荷を理解します。 
    SELECT
    wal_records, wal_fpi, wal_bytes,
    wal_buffers_full, wal_write, wal_sync,
    stats_reset
FROM pg_stat_wal;

    チェックポイント・バックグラウンドライター統計
    Checkpointerプロセスの実行状況とバックグラウンドライターの動作統計を表示します。チェックポイント完了時間（write_time, sync_time）が長い場合はI/O問題の可能性があります。 
    -- Checkpointer統計（PG18で独立）
SELECT * FROM pg_stat_checkpointer;

-- バックグラウンドライター統計
SELECT * FROM pg_stat_bgwriter;

-- Checkpoint詳細
SELECT
    num_timed, num_requested, num_done,
    write_time, sync_time, buffers_written,
    stats_reset
FROM pg_stat_checkpointer;

    ロック・トランザクション・長時間実行問題

    長時間実行トランザクション（5分以上）
    5分以上実行中のトランザクションを表示します。このようなトランザクションはVACUUMをブロックし、ディスク容量やメモリを圧迫します。 
    SELECT
    pid, application_name, state, query_start, state_change,
    age(now(), query_start) AS duration, query
FROM pg_stat_activity
WHERE state != 'idle'
    AND query_start < now() - INTERVAL '5 minutes'
ORDER BY query_start ASC;

    Idle in Transaction セッション
    アイドル状態のままトランザクションを保持しているセッションを表示します。このセッションはVACUUMをブロックするため、できるだけ早く終了する必要があります。 
    SELECT
    pid, datname, application_name, state_change,
    age(now(), state_change) AS idle_duration,
    xact_start, age(now(), xact_start) AS xact_age, query
FROM pg_stat_activity
WHERE state = 'idle in transaction'
ORDER BY xact_start ASC;

    ロック競合・待機チェーン エラー
    ロック待機中のプロセスとロック保持プロセスの関係を表示します。ブロッキング（待機）チェーンを追跡できます。 
    -- シンプルな待機チェーン表示
SELECT
    blocked.pid AS blocked_pid,
    blocked.usename AS blocked_user,
    blocked.query AS blocked_query,
    blocking.pid AS blocking_pid,
    blocking.usename AS blocking_user,
    blocking.query AS blocking_query,
    pg_blocking_pids(blocked.pid) AS blocking_pids
FROM pg_stat_activity blocked
JOIN pg_stat_activity blocking
    ON blocking.pid = ANY(pg_blocking_pids(blocked.pid))
WHERE blocked.pid != blocking.pid;

-- ロック詳細情報
SELECT
    l.pid, l.mode, l.granted, a.usename,
    a.application_name, a.query, l.locktype, l.database,
    l.relation::regclass
FROM pg_locks l
JOIN pg_stat_activity a ON l.pid = a.pid
WHERE NOT l.granted
ORDER BY l.pid;

-- 詳細なロック待機チェーン（複数段階）
SELECT
    blocked_locks.pid AS blocked_pid,
    blocked_activity.usename AS blocked_user,
    blocking_locks.pid AS blocking_pid,
    blocking_activity.usename AS blocking_user,
    blocked_activity.query AS blocked_statement,
    blocking_activity.query AS current_statement_in_blocking_process
FROM pg_catalog.pg_locks blocked_locks
JOIN pg_catalog.pg_stat_activity blocked_activity ON blocked_activity.pid = blocked_locks.pid
JOIN pg_catalog.pg_locks blocking_locks
    ON blocking_locks.locktype = blocked_locks.locktype
   AND blocking_locks.database IS NOT DISTINCT FROM blocked_locks.database
   AND blocking_locks.relation IS NOT DISTINCT FROM blocked_locks.relation
   AND blocking_locks.page IS NOT DISTINCT FROM blocked_locks.page
   AND blocking_locks.tuple IS NOT DISTINCT FROM blocked_locks.tuple
   AND blocking_locks.virtualxid IS NOT DISTINCT FROM blocked_locks.virtualxid
   AND blocking_locks.transactionid IS NOT DISTINCT FROM blocked_locks.transactionid
   AND blocking_locks.classid IS NOT DISTINCT FROM blocked_locks.classid
   AND blocking_locks.objid IS NOT DISTINCT FROM blocked_locks.objid
   AND blocking_locks.objsubid IS NOT DISTINCT FROM blocked_locks.objsubid
   AND blocking_locks.pid != blocked_locks.pid
JOIN pg_catalog.pg_stat_activity blocking_activity ON blocking_activity.pid = blocking_locks.pid
WHERE NOT blocked_locks.granted;

-- 誰かをブロックしているPIDの集合を返す：ブロックの根元を示す
SELECT pid, usename, state, wait_event_type, query, xact_start
FROM pg_stat_activity
WHERE pid IN (
    -- 誰かをブロックしているPIDの集合
    SELECT DISTINCT unnest(pg_blocking_pids(pid))
    FROM pg_stat_activity
    WHERE cardinality(pg_blocking_pids(pid)) > 0
)
AND cardinality(pg_blocking_pids(pid)) = 0;  -- 自分自身は誰にもブロックされていない

    1分以上のアクティブクエリ 警告
    1分以上実行中のアクティブなクエリを表示します。実行時間が長いクエリの最適化対象を特定できます。 
    SELECT
    pid, usename, client_addr,
    now() - xact_start AS xact_duration,
    now() - query_start AS query_duration,
    state, wait_event_type, wait_event, query
FROM pg_stat_activity
WHERE state != 'idle'
  AND (now() - query_start) > interval '1 minute'
ORDER BY query_duration DESC;

    Prepared Transactions（2PC）の確認 警告
    PREPARE TRANSACTIONで準備中のまま放置されたトランザクションを表示します。このようなトランザクションはVACUUMをブロックし、WALも蓄積されます。 
    SELECT
    gid, prepared, owner, database
FROM pg_prepared_xacts
ORDER BY prepared ASC;

    Vacuum・Autovacuum 状況
    実行中のVacuum処理の進捗、テーブル単位のDead Tuple比率、最後のVacuum実行時刻を表示します。Dead Tuple蓄積が多いテーブルは手動VACUUMまたはAutovacuum調整を検討します。 
    -- 実行中のVacuum/Autovacuum進捗
SELECT
    pid, datname, relid::regclass AS relation, phase,
    heap_blks_total, heap_blks_scanned, heap_blks_vacuumed,
    index_vacuum_count, max_dead_tuple_bytes, num_dead_item_ids,
    ROUND(100.0 * heap_blks_scanned / NULLIF(heap_blks_total, 0), 2) AS progress_percent
FROM pg_stat_progress_vacuum;

-- テーブル毎のDead Tuple比率（100K行以上のテーブルに限定）
SELECT
    schemaname, relname, n_live_tup, n_dead_tup,
    ROUND(100.0 * n_dead_tup / NULLIF(n_live_tup + n_dead_tup, 0), 2) AS dead_ratio_percent,
    last_vacuum, last_autovacuum, autovacuum_count, vacuum_count
FROM pg_stat_user_tables
WHERE n_live_tup + n_dead_tup > 100000
ORDER BY dead_ratio_percent DESC
LIMIT 20;

-- すべてのテーブルのDead Tuple数（多い順）
SELECT
    schemaname, relname, n_live_tup, n_dead_tup, n_mod_since_analyze,
    last_vacuum, last_autovacuum, last_analyze, last_autoanalyze,
    vacuum_count, autovacuum_count, analyze_count, autoanalyze_count
FROM pg_stat_user_tables
ORDER BY n_dead_tup DESC;

    Transaction ID Wraparound 監視 警告
    Transaction ID（XID）とMultiXactID（MXID）の経過を監視します。PostgreSQLのXIDは32ビット（約40億）で、ほぼ無限にはならないため定期的なVACUUMで新しいXIDを割り当てる必要があります。age値が200万を超えたらVACUUMを実行してください。 
    -- すべてのデータベースのXID age確認
SELECT
    datname,
    age(datfrozenxid) AS xid_age,
    mxid_age(datminmxid) AS mxid_age
FROM pg_database
ORDER BY age(datfrozenxid) DESC;

-- テーブル毎のXID age確認（最も古いテーブル順）
SELECT
    s.schemaname, s.relname,
    age(c.relfrozenxid) AS xid_age,
    mxid_age(c.relminmxid) AS mxid_age,
    s.n_live_tup, s.n_dead_tup, s.last_autovacuum
FROM pg_stat_user_tables s
JOIN pg_class c ON c.oid = s.relid
ORDER BY age(c.relfrozenxid) DESC
LIMIT 50;

    メモリ・リソース設定確認

    主要パラメータ設定値の確認
    メモリ、WAL、接続、チェックポイント、Autovacuum、ログなどの主要パラメータを表示します。推奨値との比較やチューニング前後の変更確認に使用します。 
    SELECT
    name, setting, unit, category, short_desc
FROM pg_settings
WHERE name IN (
    'shared_buffers', 'effective_cache_size', 'work_mem',
    'maintenance_work_mem', 'max_connections', 'max_wal_size',
    'wal_level', 'checkpoint_timeout', 'autovacuum',
    'log_min_duration_statement', 'track_commit_timestamp'
)
ORDER BY category, name;

    Streaming Replication 構成 ヘルスチェック

    Primary側の監視

    Replication Slot 状況
    物理・論理スロット、アクティブ状態、WAL保持状況、failover対応状況（PG18新機能）を表示します。各スロットの健全性を一覧で確認できます。 
    SELECT
    slot_name, slot_type, active, active_pid,
    restart_lsn, confirmed_flush_lsn, wal_status, safe_wal_size,
    inactive_since, conflicting, invalidation_reason,
    failover, synced
FROM pg_replication_slots
ORDER BY slot_name;

    Replication Slot WAL保持量の詳細
    各スロットが現在保持しているWALサイズをバイト単位と人間が読める形式で表示します。保持WAL量が多いほどStandbyが遅れているか、接続が切れていることを示唆します。 
    SELECT
    slot_name,
    slot_type,
    active,
    restart_lsn,
    pg_current_wal_lsn() AS current_lsn,
    pg_wal_lsn_diff(
        pg_current_wal_lsn(),
        restart_lsn
    ) AS retained_wal_bytes,
    pg_size_pretty(
        pg_wal_lsn_diff(
            pg_current_wal_lsn(),
            restart_lsn
        )
    ) AS retained_wal,
    wal_status,
    safe_wal_size,
    inactive_since,
    conflicting,
    invalidation_reason
FROM pg_replication_slots
WHERE restart_lsn IS NOT NULL
ORDER BY retained_wal_bytes DESC;

    Replication Lag 監視
    各Standbyとの同期状態を監視します。write_lag、flush_lag、replay_lagで遅延状況を把握します。複数のビューから異なる角度で遅延を確認できます。 
    -- Standby単位のReplication Lag
SELECT
    client_addr, usename, application_name, state,
    sync_state, sync_priority, write_lsn, flush_lsn, replay_lsn,
    write_lag, flush_lag, replay_lag, backend_xmin
FROM pg_stat_replication
ORDER BY client_addr;

-- WAL位置差分をMB単位で表示
SELECT
    client_addr, application_name,
    (pg_wal_lsn_diff(pg_current_wal_lsn(), flush_lsn) / 1024.0 / 1024.0)::numeric(10,2) AS lag_mb,
    (pg_wal_lsn_diff(pg_current_wal_lsn(), replay_lsn) / 1024.0 / 1024.0)::numeric(10,2) AS apply_lag_mb
FROM pg_stat_replication
ORDER BY lag_mb DESC;

SELECT application_name, client_addr, state, pg_current_wal_lsn() AS primary_lsn, replay_lsn,
    pg_wal_lsn_diff( pg_current_wal_lsn(), replay_lsn ) AS replay_lag_bytes,
    write_lag, flush_lag, replay_lag
FROM pg_stat_replication
ORDER BY replay_lag_bytes DESC;

    Replication Lag 詳細分析（遅延箇所の特定）
    sent_lsn → write → flush → replay の4地点のWAL位置差分を表示します。これにより、遅延がどこで発生しているかを正確に把握できます：
    • write_lag_bytes：Primary送信 → Standby WAL書き込み間のネットワーク遅延
    • flush_lag_bytes：Standby WAL書き込み → ディスク同期間のストレージ遅延
    • replay_lag_bytes：Standby WAL同期 → リプレイ完了間のApply Worker遅延
    SELECT
    pid,
    usename,
    application_name,
    client_addr,
    client_hostname,
    state,
    sent_lsn,
    write_lsn,
    flush_lsn,
    replay_lsn,
    sent_lsn - write_lsn  AS write_lag_bytes,
    sent_lsn - flush_lsn  AS flush_lag_bytes,
    sent_lsn - replay_lsn AS replay_lag_bytes,
    write_lag,
    flush_lag,
    replay_lag,
    sync_priority,
    sync_state,
    reply_time
FROM pg_stat_replication
ORDER BY application_name;

    Primary Replication接続状態
    接続中のすべてのReplication接続（Standby）の状態を監視します。返信時刻（reply_time）が古い場合は、Standbyがハングしているか遅れている可能性があります。 
    SELECT
    pid,
    usename,
    application_name,
    client_addr,
    state,
    sync_state,
    backend_start,
    reply_time,
    now() - reply_time AS reply_age
FROM pg_stat_replication
ORDER BY application_name;

    Primary WAL生成・保持状況
    WALディスク使用量、各スロットの保持WALサイズを表示します。WAL領域がいっぱいになるとレプリケーションが停止します。 
    SELECT
    pg_current_wal_lsn() AS current_lsn,
    pg_wal_lsn_diff(pg_current_wal_lsn(), '0/0'::pg_lsn) / 1024.0 / 1024.0 / 1024.0 AS total_wal_gb,
    pg_size_pretty(CAST((SELECT sum(size) FROM pg_ls_waldir()) AS int8)) AS wal_dir_size,
    (SELECT setting FROM pg_settings WHERE name = 'wal_keep_size') AS wal_keep_size;

-- スロット毎のWAL保持サイズ
SELECT
    slot_name, slot_type, restart_lsn,
    pg_wal_lsn_diff(pg_current_wal_lsn(), restart_lsn) / 1024.0 / 1024.0 / 1024.0 AS retained_wal_gb
FROM pg_replication_slots
WHERE slot_type = 'physical'
ORDER BY retained_wal_gb DESC;

    スロット無効化・WAL喪失の検知
    Standbyが追いつけずスロットが失効状態になった場合を検知します。wal_status='lost'の場合はスロット無効化されており、Standbyの初期化が必要です。 
    SELECT
    slot_name, slot_type, wal_status, invalidation_reason,
    restart_lsn, safe_wal_size,
    pg_size_pretty(safe_wal_size) AS safe_wal_size_pretty
FROM pg_replication_slots
WHERE wal_status != 'reserved'
   OR invalidation_reason IS NOT NULL;

    Replication Lag上限超過検知
    許容値を超えたReplication Lagを自動検知します。CRITICAL/WARNING/OKで段階分けしています。 
    SELECT
    client_addr, application_name, state, replay_lag,
    CASE WHEN replay_lag > INTERVAL '5 seconds' THEN 'CRITICAL'
         WHEN replay_lag > INTERVAL '1 second' THEN 'WARNING'
         ELSE 'OK' END AS severity
FROM pg_stat_replication
WHERE replay_lag > INTERVAL '1 second'
ORDER BY replay_lag DESC;

    Standby側の監視

    Recovery状態・進捗確認
    Standbyの復旧（Redo）状況を確認します。最後に受信・リプレイされたLSN位置とリプレイ遅延時間を監視します。 
    SELECT
    pg_is_in_recovery() AS is_standby,
    pg_last_wal_receive_lsn() AS last_received_lsn,
    pg_last_wal_replay_lsn() AS last_replayed_lsn,
    pg_is_wal_replay_paused() AS is_paused,
    (CASE WHEN pg_is_in_recovery()
           THEN now() - pg_postmaster_start_time()
           ELSE NULL END)::interval(0) AS recovery_uptime;

SELECT
    pg_is_in_recovery() AS in_recovery,
    pg_last_wal_receive_lsn() AS receive_lsn,
    pg_last_wal_replay_lsn() AS replay_lsn,
    pg_last_xact_replay_timestamp() AS last_replay_time,
    now() - pg_last_xact_replay_timestamp() AS replay_delay;

    WAL Receiver状態
    WAL Receiverプロセスの詳細状態を表示します。WAL受信、書き込み、フラッシュの進捗状況を監視します。status='streaming'で正常に接続中を示します。 
    SELECT
    pid,
    status,
    receive_start_lsn,
    receive_start_tli,
    written_lsn,
    flushed_lsn,
    latest_end_lsn,
    latest_end_time,
    slot_name,
    sender_host,
    sender_port,
    conninfo
FROM pg_stat_wal_receiver;

SELECT
    status,
    written_lsn,
    flushed_lsn,
    latest_end_lsn,
    latest_end_time,
    slot_name,
    sender_host,
    sender_port
FROM pg_stat_wal_receiver;

    Standby接続検証
    Standbyがprimaryに正常に接続できているか確認します。walreceiverプロセスがactiveならば接続成功です。 
    -- Standby側で実行
SELECT
    pid, usename, application_name, state,
    query_start, backend_start, query
FROM pg_stat_activity
WHERE application_name = 'walreceiver'
    OR (state = 'active' AND query LIKE 'START_REPLICATION%');

    Recovery Conflicts検知
    Standby側で発生したRecovery Conflicts（リプレイ中のクエリ競合）を監視します。Standbyで実行中のクエリがWALリプレイと競合した場合、クエリが中断される可能性があります。各種競合タイプをカウントします。 
    SELECT
    datname,
    confl_tablespace,
    confl_lock,
    confl_snapshot,
    confl_bufferpin,
    confl_deadlock
FROM pg_stat_database_conflicts
ORDER BY datname;
